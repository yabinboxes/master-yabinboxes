Publish by API

Use your CI system to call the publish endpoint.

GitHub Action example
name: Publish by API
on:
  push:
    branches: [main]

jobs:
  publish:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Publish to Snow
        env:
          SNOW_API_URL: https://your-api.example.com/connectors/publish-snow-deployment
          SNOW_TOKEN: ${{ secrets.SNOW_TOKEN }}
        run: |
          curl -X POST "$SNOW_API_URL" \
            -H "Authorization: Bearer $SNOW_TOKEN" \
            -H "Content-Type: application/json" \
            -d '{
              "snowName": "${{ github.repository }}",
              "registry": "github",
              "repositoryOwner": "${{ github.repository_owner }}",
              "url": "${{ github.server_url }}/${{ github.repository }}"
            }'
